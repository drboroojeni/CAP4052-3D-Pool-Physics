cmake_minimum_required(VERSION 3.21)

# Avoid requiring a newer CMake for extraction timestamps.
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
# Share one static runtime setting across the app and its dependencies.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING
    "MSVC runtime used by the project and its static dependencies")
project(CAP4052_DF5_Pool VERSION 1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_RENDERER "Build the interactive GLFW/OpenGL demonstration" ON)
# The normal Windows build exposes the game, with tests explicitly opt-in.
if(WIN32)
  option(BUILD_TESTING "Build the optional physics tests" OFF)
else()
  option(BUILD_TESTING "Build the optional physics tests" ON)
endif()
include(CTest)

add_library(pool_physics STATIC src/Physics.cpp)
target_include_directories(pool_physics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
  target_compile_options(pool_physics PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(pool_physics PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
  add_executable(pool_physics_tests tests/physics_tests.cpp)
  target_link_libraries(pool_physics_tests PRIVATE pool_physics)
  add_test(NAME pool_physics_tests COMMAND pool_physics_tests)
  add_executable(pool_feedback_tests tests/feedback_tests.cpp)
  target_link_libraries(pool_feedback_tests PRIVATE pool_physics)
  add_test(NAME pool_feedback_tests COMMAND pool_feedback_tests)
endif()

if(BUILD_RENDERER)
  find_package(OpenGL REQUIRED)
  include(cmake/Dependencies.cmake)
  add_library(glad STATIC third_party/glad/src/glad.c)
  target_include_directories(glad PUBLIC third_party/glad/include)
  target_link_libraries(glad PUBLIC ${CMAKE_DL_LIBS})

  find_package(Threads REQUIRED)
  add_library(pool_audio STATIC src/CollisionAudio.cpp)
  target_include_directories(pool_audio PUBLIC src)
  target_include_directories(pool_audio SYSTEM PRIVATE third_party/miniaudio)
  target_link_libraries(pool_audio PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
  if(MSVC)
    target_compile_options(pool_audio PRIVATE /utf-8)
  endif()

  add_executable(CAP4052_DF5_Pool src/main.cpp src/Renderer3D.cpp)
  target_link_libraries(CAP4052_DF5_Pool PRIVATE pool_physics pool_audio glad glfw glm::glm OpenGL::GL)
  target_compile_definitions(CAP4052_DF5_Pool PRIVATE GLFW_INCLUDE_NONE)
  set_target_properties(CAP4052_DF5_Pool PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(CAP4052_DF5_Pool PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(CAP4052_DF5_Pool PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
